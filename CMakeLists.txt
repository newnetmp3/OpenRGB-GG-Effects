cmake_minimum_required(VERSION 3.16)
project(OpenRGBGGEffects LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ggfx INTERFACE)
target_include_directories(ggfx INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
enable_testing()
add_executable(ggfx_test tests/effects_test.cpp)
target_link_libraries(ggfx_test PRIVATE ggfx)
add_test(NAME ggfx_core COMMAND ggfx_test)
